// The shared NIMS WebView shim is now valid source and no longer requires
// line-number based post-processing. Keep this task hook as a deterministic
// validation step so generated assets cannot silently reintroduce the removed
// runtime symbols that previously broke login/navigation.
tasks.named('syncNimsRuntimeAssets').configure {
    doLast {
        def directory = layout.buildDirectory.dir('generated/nimsRuntimeAssets').get().asFile
        def runtimeFile = new File(directory, 'nimsWebviewShim.js')
        assert runtimeFile.isFile()
        def source = runtimeFile.getText('UTF-8')
        assert source.contains('__nimsInjectedAt')
        assert source.contains('function patchOffset()')
        assert source.contains('function patchAjaxCompleteTab()')
        assert !source.contains('postRuntime(')
        assert !source.contains('reportReady(')
        assert !source.contains('phase ||')
    }
}
