tasks.named("syncNimsRuntimeAssets").configure {
    doLast {
        val directory = layout.buildDirectory.dir("generated/nimsRuntimeAssets").get().asFile
        val runtimeFile = directory.listFiles().orEmpty()
            .filter { it.extension == "js" }
            .sortedBy { it.name }
            .last()
        val lines = runtimeFile.readLines().toMutableList()
        check(lines.size >= 190)
        lines[163] = "          return undefined;"
        repeat(2) { lines.removeAt(164) }
        lines[179] = "    } catch (error) { /* compatibility guard */ }"
        repeat(3) { lines.removeAt(180) }
        runtimeFile.writeText(lines.joinToString("\n", postfix = "\n"))
    }
}
